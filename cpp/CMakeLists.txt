cmake_minimum_required(VERSION 3.16)
project(abstraction_rights VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(NOT TARGET abstraction::ipc)
 find_package(abstraction_ipc CONFIG QUIET)
endif()
if(NOT TARGET abstraction::ipc AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../abstraction-identity/cpp/CMakeLists.txt")
 add_subdirectory(../../abstraction-identity/cpp abstraction_ipc-build)
endif()
if(NOT TARGET abstraction::ipc)
 message(FATAL_ERROR "Install abstraction_ipc or provide sibling abstraction-identity. Nothing is fetched.")
endif()
add_library(abstraction_rights_client INTERFACE)
add_library(abstraction::rights_client ALIAS abstraction_rights_client)
set_target_properties(abstraction_rights_client PROPERTIES EXPORT_NAME rights_client)
target_compile_features(abstraction_rights_client INTERFACE cxx_std_17)
target_include_directories(abstraction_rights_client INTERFACE
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
 $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(abstraction_rights_client INTERFACE abstraction::ipc)
install(TARGETS abstraction_rights_client EXPORT abstraction_rightsTargets)
install(DIRECTORY abstraction/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/abstraction)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
set(config_dir ${CMAKE_INSTALL_LIBDIR}/cmake/abstraction_rights)
install(EXPORT abstraction_rightsTargets NAMESPACE abstraction:: DESTINATION ${config_dir})
configure_package_config_file(cmake/abstraction_rightsConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/abstraction_rightsConfig.cmake INSTALL_DESTINATION ${config_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/abstraction_rightsConfig.cmake DESTINATION ${config_dir})
